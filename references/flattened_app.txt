# 15     def create(cls, id: int, **kwargs):
# 16         #  Step 1: Load config from DB
# 17         engine = create_engine(f"sqlite:///{DB_PATH}")
# 18         with Session(bind=engine) as session:
# 19             config = session.get(cls.config_model, id)

# 35                 provider_class = obj

# 39             raise ImportError(f"No valid subclass of {cls.base_class.__name__} found in {ext_path}")
