# 14 class LintingScoreProvider(ScoreProviderBase):
# 15     """
# 16     Aggregates Ruff, Black and Mypy tool scores into a single linting metric.
# 17

# 31     _DEFAULT_WEIGHTS: Dict[str, float] = {"ruff": 0.7, "black": 0.2, "mypy": 0.1}

# 35     #  ───────────────────────────────────────────────────────────── core entry
# 36     def _run(self, input: dict) -> ScoreOutputSchema:  #  noqa: D401, N802

# 39         #  1 ─────────────────────────────────────────── resolve & strip the code
# 40         file_path = self._make_path_from_raw(file_path)
# 41         full_code = file_path.read_text(encoding="utf-8")
# 42         clean_code, _ = split_code_and_notes(full_code)

# 44         stripped_path = self._write_working_copy(clean_code)
# 45 
# 46         #  2 ─────────────────────────────────────────────── tool execution
# 47         available = {tp._config.name.lower(): tp for tp in self.tool_providers}
# 48         ruff_score, ruff_violations, tool_failures = self._run_ruff(available, stripped_path)
# 49         black_score, _ = self._run_tool("black", available, stripped_path, tool_failures)
# 50         mypy_score, _ = self._run_tool("mypy", available, stripped_path, tool_failures)
# 51 
# 52         #  3 ─────────────────────────────────────────── weighting & threshold
# 53         weights = self._parse_weights(input.get("weights"))
# 54         weighted_score = round(
# 55             ruff_score * weights["ruff"]
# 56             + black_score * weights["black"]
# 57             + mypy_score * weights["mypy"],
# 58             3,
# 59         )
# 60 
# 61         threshold = input.get("threshold", self._DEFAULT_THRESHOLD)
# 62         meets_threshold = (threshold is None) or (weighted_score >= threshold)
# 63 
# 64         #  4 ───────────────────────────────────────────── components / summary
# 65         components: Dict[str, float] = {
# 66             "ruff_score": ruff_score,
# 67             "ruff_violations": float(len(ruff_violations)),
# 68             "black_score": black_score,
# 69             "mypy_score": mypy_score,
# 70             "tool_failure_count": float(len(tool_failures)),
# 71             "weight_ruff": weights["ruff"],
# 72             "weight_black": weights["black"],
# 73             "weight_mypy": weights["mypy"],
# 74         }
# 75         if threshold is not None:
# 76             components["threshold"] = threshold
# 77             components["meets_threshold"] = 1.0 if meets_threshold else 0.0
# 78 
# 79         summary = self._build_summary(
# 80             ruff_violations=ruff_violations,
# 81             tool_failures=tool_failures,
# 82             meets_threshold=meets_threshold,
# 83             threshold=threshold,
# 84         )

# 112     #  2️⃣  TOOL RUNNERS ----------------------------------------------------
# 113     def _run_ruff(
# 114         self,
# 115         available: Dict[str, Any],
# 116         target_path: Path,
# 117     ) -> Tuple[float, List[str], Dict[str, str]]:
# 118         """Runs Ruff and returns (score, violations, failures‑dict)."""
# 119         tool_failures: Dict[str, str] = {}
# 120 
# 121         score, violations = self._run_tool(
# 122             "ruff", available, target_path, tool_failures, collect_violations=True
# 123         )
# 124 
# 125         #  Ruff score: scaled linearly down by violation count, capped at 10+
# 126         score = 1.0 - min(1.0, len(violations) / self._MAX_RUFF_VIOLATIONS_CONSIDERED)
# 127         return score, violations, tool_failures
# 128 
# 129 
# 130     def _run_tool(
# 131         self,
# 132         name: str,
# 133         available: Dict[str, Any],
# 134         target_path: Path,
# 135         failures: Dict[str, str],
# 136         collect_violations: bool = False,
# 137     ) -> Tuple[float, List[str]]:
# 138         """
# 139         Returns a score in [0, 1] based on violation count if tool succeeded,
# 140         or 0.0 if the tool failed (e.g., crashed or missing).
# 141         """
# 142         if name not in available:
# 143             failures[name] = "not_available"
# 144             return 0.0, []
# 145 
# 146         try:
# 147             result = available[name].run({"target": str(target_path), "check": True})
# 148         except Exception as exc:
# 149             failures[name] = f"crashed: {exc}"
# 150             return 0.0, []
# 151 
# 152         #  Always try to extract violations
# 153         violations = getattr(result, "violations", []) or []
# 154         rc = getattr(result, "return_code", 0)
# 155 
# 156         #  Only fail hard if the tool couldn't run
# 157         if rc not in {0, 1} and not violations:
# 158             failures[name] = result.summary or "failed"
# 159             return (0.0, []) if not collect_violations else (0.0, violations)
# 160 
# 161         #  Compute score based on violation count
# 162         count = len(violations)
# 163         if name == "ruff":
# 164             score = 1.0  #  Ruff handled in _run_ruff
# 165         else:
# 166             score = 1.0 - min(1.0, 0.1 * count)
# 167 
# 168         return (score, violations) if collect_violations else (score, [])
# 169 
# 170 
# 171     #  3️⃣  WEIGHT PARSING --------------------------------------------------
# 172     def _parse_weights(self, custom: Dict[str, float] | None) -> Dict[str, float]:
# 173         weights = {**self._DEFAULT_WEIGHTS}
# 174 
# 175         if custom:
# 176             weights.update(
# 177                 {k.lower(): float(v) for k, v in custom.items() if k.lower() in weights}
# 178             )
# 179 
# 180         total = sum(weights.values()) or 1.0
# 181         return {k: v / total for k, v in weights.items()}
# 182 
# 183     #  4️⃣  SUMMARY BUILDER --------------------------------------------------
# 184     def _build_summary(
# 185         self,
# 186         *,
# 187         ruff_violations: List[str],

# 189         meets_threshold: bool,
# 190         threshold: float | None,
# 191     ) -> str:
# 192         lines: List[str] = []
# 193 
# 194         if ruff_violations:
# 195             top_codes = {}
# 196             for code in ruff_violations:
# 197                 top_codes[code] = top_codes.get(code, 0) + 1
# 198             top_sorted = sorted(top_codes.items(), key=lambda kv: kv[1], reverse=True)
# 199             lines.append(
# 200                 "Top Ruff violations: "
# 201                 + ", ".join(f"{c}×{n}" for c, n in top_sorted[:3])
# 202             )
# 203         else:
# 204             lines.append("No Ruff violations detected")
# 205 
# 206         if tool_failures:
# 207             lines.append("⚠️  Tool failures: " + ", ".join(tool_failures))
# 208 
# 209         if threshold is not None:
# 210             state = "✅ meets" if meets_threshold else "❌ below"
# 211             lines.append(f"{state} threshold {threshold:.2f}")
# 212 
# 213         return "\n".join(lines)
