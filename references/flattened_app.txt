# 8 
# 9 
# 10 class BaseProviderFactory:
# 11     config_model = None  
# 12     base_class = BaseProvider  
# 13 
# 14     @classmethod
# 15     def create(cls, id: int, **kwargs):
# 16         #  Step 1: Load config from DB
# 17         engine = create_engine(f"sqlite:///{DB_PATH}")
# 18         with Session(bind=engine) as session:
# 19             config = session.get(cls.config_model, id)
# 20             if not config:
# 21                 raise ValueError(f"{cls.config_model.__name__} ID {id} not found")
# 22 
# 23         #  Step 2: Load extension
# 24         ext_path = Path("extensions") / f"{config.artifact_path}.py"
# 25         if not ext_path.exists():
# 26             raise FileNotFoundError(f"Extension not found: {ext_path}")
# 27 
# 28         spec = spec_from_file_location("provider_extension", ext_path)
# 29         module = module_from_spec(spec)
# 30         spec.loader.exec_module(module)
# 31 
# 32         provider_class = None
# 33         for obj in module.__dict__.values():
# 34             if isinstance(obj, type) and issubclass(obj, cls.base_class) and obj is not cls.base_class:
# 35                 provider_class = obj
# 36                 break
# 37 
# 38         if not provider_class:
# 39             raise ImportError(f"No valid subclass of {cls.base_class.__name__} found in {ext_path}")
# 40 
# 41         #  Step 3: Instantiate and wire dependencies
# 42         instance = provider_class(config=config)
# 43         instance.resolve_dependencies(**kwargs)
# 44         return instance
