# 152     def _run_tool_with_context(

# 165         if rc not in {0, 1} and not violations:
# 166             return 0.0, violations

# 30         tree = ast.parse(source)
# 31         symbols = set()
# 32 
# 33         for node in ast.walk(tree):

# 35                 metrics["function_count"] += 1

# 42                 if ast.get_docstring(node):
# 43                     metrics["docstring_count"] += 1
# 44             elif isinstance(node, (ast.If, ast.For, ast.While, ast.Match, ast.With, ast.Try)):
# 45                 metrics["branch_count"] += 1
# 46             elif isinstance(node, ast.Call):
# 47                 metrics["call_count"] += 1
# 48             elif isinstance(node, ast.Assign):
# 49                 metrics["assignment_count"] += 1
# 50             elif isinstance(node, ast.Return):

# 52             elif isinstance(node, (ast.Try, ast.ExceptHandler, ast.Raise)):
# 53                 metrics["exception_count"] += 1
