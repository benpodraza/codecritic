# 9 def analyze_code(source: str) -> Dict[str, Any]:
# 10     """Analyze Python source code and return structural metrics."""
# 11     metrics: Dict[str, Any] = {}
# 12 
# 13     #  === Basic line count ===
# 14     lines = source.strip().splitlines()
# 15     metrics["line_count"] = len(lines)
# 16 
# 17     #  === AST-based metrics ===
# 18     tree = ast.parse(source)
# 19     symbols = set()
# 20 
# 21     metrics.update({
# 22         "function_count": 0,
# 23         "class_count": 0,
# 24         "branch_count": 0,
# 25         "call_count": 0,
# 26         "assignment_count": 0,
# 27         "return_count": 0,
# 28         "exception_count": 0,
# 29         "docstring_count": 0,
# 30     })
# 31 
# 32     for node in ast.walk(tree):
# 33         if isinstance(node, ast.FunctionDef):
# 34             metrics["function_count"] += 1
# 35             symbols.add(node.name)
# 36             if ast.get_docstring(node):
# 37                 metrics["docstring_count"] += 1
# 38         elif isinstance(node, ast.ClassDef):
# 39             metrics["class_count"] += 1
# 40             symbols.add(node.name)
# 41             if ast.get_docstring(node):
# 42                 metrics["docstring_count"] += 1
# 43         elif isinstance(node, (ast.If, ast.For, ast.While, ast.Match, ast.With, ast.Try)):
# 44             metrics["branch_count"] += 1
# 45         elif isinstance(node, ast.Call):
# 46             metrics["call_count"] += 1
# 47         elif isinstance(node, ast.Assign):
# 48             metrics["assignment_count"] += 1
# 49         elif isinstance(node, ast.Return):
# 50             metrics["return_count"] += 1
# 51         elif isinstance(node, (ast.Try, ast.ExceptHandler, ast.Raise)):
# 52             metrics["exception_count"] += 1
# 53 
# 54     metrics["symbol_count"] = len(symbols)
# 55 
# 56     #  === Token-based metrics ===
# 57     comment_count = 0
# 58     import_count = 0
# 59     try:
# 60         tokens = tokenize.generate_tokens(StringIO(source).readline)
# 61         for toknum, tokval, *_ in tokens:
# 62             if toknum == tokenize.COMMENT:
# 63                 comment_count += 1
# 64             elif toknum == tokenize.NAME and tokval in {"import", "from"}:
# 65                 import_count += 1
# 66     except tokenize.TokenError:
# 67         pass
# 68 
# 69     metrics["comment_count"] = comment_count
# 70     metrics["import_count"] = import_count
# 71
